Usage
<!-- With label (recommended) -->
<label class="mdst-checkbox-label">
  <input type="checkbox" class="mdst-checkbox" />
  Label text
</label>

<!-- Standalone -->
<input type="checkbox" class="mdst-checkbox" />

<!-- States -->
<input type="checkbox" class="mdst-checkbox" checked />
<input type="checkbox" class="mdst-checkbox" disabled />

<!-- Toggle -->
<input type="checkbox" class="mdst-toggle" />
<input type="checkbox" class="mdst-toggle" checked />
<input type="checkbox" class="mdst-toggle" disabled />

<!-- Toggle rounded -->
<input type="checkbox" class="mdst-toggle mdst-toggle--pill" />
<input type="checkbox" class="mdst-toggle mdst-toggle--pill" checked />
