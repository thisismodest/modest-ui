Usage
<!-- With label (recommended) -->
<label>
  <input type="checkbox" class="mdst-input" />
  Label text
</label>

<!-- Standalone -->
<input type="checkbox" class="mdst-input" />

<!-- States -->
<input type="checkbox" class="mdst-input" checked />
<input type="checkbox" class="mdst-input" disabled />

<!-- Toggle -->
<input type="checkbox" class="mdst-input" role="switch" />
<input type="checkbox" class="mdst-input" role="switch" checked />
<input type="checkbox" class="mdst-input" role="switch" disabled />

<!-- Toggle rounded -->
<input type="checkbox" class="mdst-input mdst-input--pill" role="switch" />
<input type="checkbox" class="mdst-input mdst-input--pill" role="switch" checked />
