Drag-and-drop included. The native <input type="file"> supports dragging files onto it out of the box. The button is styled via ::file-selector-button and the filename updates automatically on selection.

Usage
<!-- Basic file input -->
<input type="file" class="mdst-input" />

<!-- Borderless -->
<input type="file" class="mdst-input mdst-input--borderless" />

<!-- Solid button -->
<input type="file" class="mdst-input mdst-input--solid" />

<!-- Disabled -->
<input type="file" class="mdst-input" disabled />

<!-- Accept only images -->
<input type="file" class="mdst-input" accept="image/*,.png,.jpg,.jpeg,.gif,.webp" />

<!-- Multiple files -->
<input type="file" class="mdst-input" multiple />
